import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="One list, five agents" icon="open-book">
    Every session from every agent's own store, in a table, a terminal UI or a local web UI. Rename, move, archive, delete and export through each agent's sanctioned path — never by guessing at its files.
  </Card>
  <Card title="Search everything" icon="magnifier">
    Full-text search over every message of every session, including subagent transcripts, from an incremental index in asm's own directory.
  </Card>
  <Card title="Carry a conversation across agents" icon="random">
    `asm import` converts through a documented IR into a native session the target agent resumes itself — with a loss report that says what could not cross.
  </Card>
  <Card title="Sync across machines" icon="laptop">
    A hub archives sessions; machines push and pull them under their original ids. `claude --resume <id>` works on the second laptop. A background daemon keeps the hub current.
  </Card>
</CardGrid>
